# Calculate Expiration and Status dynamically
today = datetime.now().date()
def evaluate_status(date_str):
    try:
        emision = datetime.strptime(str(date_str), "%Y-%m-%d").date()
        caducidad = emision + timedelta(days=365)
        dias_restantes = (caducidad - today).days
        if dias_restantes < 0:
            return "Caducado 🔴"
        elif dias_restantes <= 30:
            return "Por Vencer (30 días) 🟡"
        else:
            return "Vigente 🟢"
    except:
        return "Desconocido"

df['fecha_caducidad'] = pd.to_datetime(df['fecha_emision']).dt.date.apply(lambda x: x + timedelta(days=365) if pd.notnull(x) else None)
df['estado_vigencia'] = df['fecha_emision'].apply(evaluate_status)
